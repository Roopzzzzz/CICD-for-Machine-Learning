pandas scikit-learn joblib import gradio as gr import pandas as pd import joblib from pyodide.http import pyfetch # TODO: Replace YOUR_HF_USERNAME with your actual Hugging Face profile username MODEL_URL = ( "https://huggingface.co" "YOUR_HF_USERNAME/drug-classification-lite/" "resolve/main/model/drug_model.joblib" ) # Download model binary directly into the browser's virtual filesystem response = await pyfetch(MODEL_URL) if response.status != 200: raise RuntimeError(f"Model download failed: HTTP {response.status}") model_bytes = await response.bytes() with open("/drug_model.joblib", "wb") as model_file: model_file.write(model_bytes) # Load the Scikit-learn preprocessing + evaluation pipeline model = joblib.load("/drug_model.joblib") def predict_drug(age, sex, bp, cholesterol, na_to_k): if age is None or na_to_k is None: return "Enter all required numerical values.", "" if age < 0 or na_to_k < 0: return "Values must be non-negative.", "" sample = pd.DataFrame([{ "Age": age, "Sex": sex, "BP": bp, "Cholesterol": cholesterol, "Na_to_K": na_to_k }]) prediction = model.predict(sample)[0] confidence = "" if hasattr(model, "predict_proba"): probabilities = model.predict_proba(sample)[0] confidence = f"{max(probabilities) * 100:.2f}%" return str(prediction), confidence demo = gr.Interface( fn=predict_drug, inputs=[ gr.Number(label="Age", value=30), gr.Dropdown(choices=["F", "M"], value="F", label="Sex"), gr.Dropdown(choices=["LOW", "NORMAL", "HIGH"], value="NORMAL", label="Blood Pressure Category"), gr.Dropdown(choices=["NORMAL", "HIGH"], value="NORMAL", label="Cholesterol Category"), gr.Number(label="Na_to_K Ratio", value=15.0) ], outputs=[ gr.Textbox(label="Predicted Class"), gr.Textbox(label="Model Confidence") ], title="Drug Classification", description="Educational ML demonstration using a trained Scikit-learn pipeline executing serverless in the browser." ) demo.launch()
